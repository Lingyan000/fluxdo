import 'dart:async';
import 'package:flutter/services.dart';
import 'package:flutter/widgets.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:fluxdo_plugin_runtime/fluxdo_plugin_runtime.dart';
import 'package:plugin_mobile_smoke/script_engine.dart';

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  testWidgets('设备执行查询、分页与隔离销毁', (tester) async {
    await tester.pumpWidget(const Directionality(textDirection:TextDirection.ltr,child:Center(child:Text('Plugin runtime smoke'))));
    final engine=await IsolateScriptEngine.spawn();
    final done=Completer<Map>();
    engine.onMessage=(value){if(value is Map&&value['t']=='smoke')done.complete(value);};
    engine.onFatal=(reason){if(!done.isCompleted)done.completeError(StateError(reason));};
    try {
      await engine.evaluate(metaversePluginPrelude);
      final data=(await rootBundle.loadString('assets/data.mjs')).replaceAll(RegExp(r'^export ',multiLine:true),'');
      await engine.evaluate('''
        $data
        (async()=>{
          const query=createQueryClient();let reads=0;
          const result=await Promise.all([query.query('k',async()=>++reads),query.query('k',async()=>++reads)]);
          const pages=createPagedList({itemKey:x=>x,load:async()=>({items:[1,2],nextCursor:null})});
          await pages.load();query.dispose();pages.dispose();
          sendMessage('${MetaverseScriptEngine.channel}',JSON.stringify({t:'smoke',reads,count:pages.state.items.length,result}));
        })();
      ''');
      final result=await done.future.timeout(const Duration(seconds:10));
      expect(result['reads'],1);expect(result['count'],2);expect(result['result'],[1,1]);
    } finally {engine.dispose();await engine.terminated.timeout(const Duration(seconds:5));}
  });
}
